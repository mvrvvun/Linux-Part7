
# 📦 dpkg (Debian packages)
dpkg -s vlc                # Check if VLC is installed (status)
dpkg --info package.deb    # Show information about a .deb package
dpkg -l vlc                # List details of installed VLC
dpkg -i package.deb        # Install a .deb package
dpkg -r vlc                # Remove package (keep config files)
dpkg -P vlc                # Purge package (remove completely incl. configs)
dpkg -c package.deb        # List files inside a .deb package

wget <binary-url>          # Download a package (.deb) from the internet

# =========================
# 📦 APT (Advanced Packaging Tool)
# =========================
/etc/apt/sources.list      # File where apt repositories are stored

apt list -a app            # Show all available versions of a package
apt list                   # List all packages
apt list --installed       # List installed packages
apt list --installed | wc -l   # Count installed packages

apt search app             # Search for a package
apt show app               # Show package details

sudo apt install app       # Install a package
sudo apt download app      # Download package (without installing)
sudo apt remove app        # Remove package (keep configs)
sudo apt purge app         # Remove package completely (with configs)

# =========================
# 💽 Disk & Storage Commands
# =========================
df -h                      # Show disk usage of all mounted partitions
du -sh *                   # Show size of all items in current folder
du -sh                     # Show total size of current folder
lsblk                      # List all block devices (partitions/disks)
sudo fdisk -l              # Show detailed partition table (all disks)
